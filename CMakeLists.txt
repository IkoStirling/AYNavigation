cmake_minimum_required(VERSION 3.25)
project(AYNavigation VERSION 0.4.0 LANGUAGES CXX)
include(CTest)

add_library(AYNavigationCore STATIC src/Graph.cpp src/Following.cpp src/Avoidance.cpp
    src/Navigator.cpp src/StaticBake.cpp src/Perception.cpp)
add_library(AYNavigation::Core ALIAS AYNavigationCore)
target_compile_features(AYNavigationCore PUBLIC cxx_std_23)
target_include_directories(AYNavigationCore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
find_package(nlohmann_json CONFIG QUIET)
if(nlohmann_json_FOUND)
    add_library(AYNavigationPerceptionJson STATIC src/PerceptionJson.cpp)
    add_library(AYNavigation::PerceptionJson ALIAS AYNavigationPerceptionJson)
    target_link_libraries(AYNavigationPerceptionJson PUBLIC AYNavigationCore
        PRIVATE nlohmann_json::nlohmann_json)
    target_compile_features(AYNavigationPerceptionJson PUBLIC cxx_std_23)
    if(MSVC)
        target_compile_options(AYNavigationPerceptionJson PRIVATE /W4 /utf-8)
    endif()
endif()
if(MSVC)
    target_compile_options(AYNavigationCore PRIVATE /W4 /utf-8)
else()
    target_compile_options(AYNavigationCore PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(AY_NAVIGATION_BUILD_AUTHORING "Build explicit .nvg graph file authoring" ON)
if(AY_NAVIGATION_BUILD_AUTHORING)
    add_library(AYNavigationAuthoring STATIC src/GraphAsset.cpp)
    add_library(AYNavigation::Authoring ALIAS AYNavigationAuthoring)
    target_link_libraries(AYNavigationAuthoring PUBLIC AYNavigationCore)
endif()

option(AY_NAVIGATION_BUILD_BEHAVIOR "Build optional AYBehavior integration" ${AY_ENABLE_BEHAVIOR})
option(AY_NAVIGATION_BUILD_PHYSICS "Build optional physics capsule motor and real jump traversal" OFF)
if(AY_NAVIGATION_BUILD_PHYSICS)
    if(NOT TARGET AYPhysics)
        message(FATAL_ERROR "AY_NAVIGATION_BUILD_PHYSICS requires AYPhysics")
    endif()
    add_library(AYNavigationPhysics STATIC src/CharacterMotor.cpp)
    add_library(AYNavigation::Physics ALIAS AYNavigationPhysics)
    target_link_libraries(AYNavigationPhysics PUBLIC AYNavigationCore AYPhysics)
endif()
option(AY_NAVIGATION_BUILD_WORLD_ADAPTER "Build optional Scene-World geometry and physics-query adapter" OFF)
if(AY_NAVIGATION_BUILD_WORLD_ADAPTER)
    if(NOT TARGET AYEntityCore OR NOT TARGET AYMath)
        message(FATAL_ERROR "Navigation World adapter requires AYEntityCore and AYMath")
    endif()
    add_library(AYNavigationWorldAdapter STATIC src/WorldBake.cpp src/PhysicsVisibility.cpp
        src/ConnectedSpaceSound.cpp)
    add_library(AYNavigation::WorldAdapter ALIAS AYNavigationWorldAdapter)
    target_link_libraries(AYNavigationWorldAdapter PUBLIC AYNavigationCore AYEntityCore AYMath)
    target_include_directories(AYNavigationWorldAdapter PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/../AYPhysics/interface
        ${CMAKE_CURRENT_SOURCE_DIR}/../AYPhysics/include)
    add_executable(AYNavigationWorldAdapterExample examples/WorldSpatialAwareness.cpp)
    target_link_libraries(AYNavigationWorldAdapterExample PRIVATE AYNavigationWorldAdapter)
endif()
if(AY_NAVIGATION_BUILD_BEHAVIOR)
    if(NOT TARGET AYBehaviorCore)
        message(FATAL_ERROR "Navigation behavior integration requires AYBehaviorCore")
    endif()
    add_library(AYNavigationBehavior STATIC src/BehaviorIntegration.cpp)
    add_library(AYNavigation::Behavior ALIAS AYNavigationBehavior)
    target_link_libraries(AYNavigationBehavior PUBLIC AYNavigationCore AYBehaviorCore)
endif()
set(_navigation_entity_default OFF)
if(TARGET AYNavigationBehavior AND TARGET AYBehaviorEntity)
    set(_navigation_entity_default ON)
endif()
option(AY_NAVIGATION_BUILD_ENTITY "Build optional root kinematic Entity motor" ${_navigation_entity_default})
if(AY_NAVIGATION_BUILD_ENTITY)
    if(NOT TARGET AYNavigationBehavior OR NOT TARGET AYBehaviorEntity)
        message(FATAL_ERROR "Navigation Entity motor requires NavigationBehavior and AYBehaviorEntity")
    endif()
    add_library(AYNavigationEntity STATIC src/EntityMotor.cpp)
    add_library(AYNavigation::Entity ALIAS AYNavigationEntity)
    target_link_libraries(AYNavigationEntity PUBLIC AYNavigationBehavior AYBehaviorEntity)
endif()
option(AY_NAVIGATION_BUILD_EXAMPLES "Build the Windows path-authoring navigation laboratory" ON)
if(TARGET AYNavigationEntity AND TARGET AYNavigationAuthoring AND (BUILD_TESTING OR AY_NAVIGATION_BUILD_EXAMPLES))
    add_library(AYNavigationLab STATIC examples/WorldLab.cpp)
    target_include_directories(AYNavigationLab PUBLIC examples)
    target_link_libraries(AYNavigationLab PUBLIC AYNavigationEntity AYNavigationAuthoring AYBehaviorInteraction)
endif()
if(AY_NAVIGATION_BUILD_EXAMPLES AND WIN32)
    if(TARGET AYNavigationLab)
        add_executable(AYNavigationWorldDemo WIN32 examples/WorldWindow.cpp)
        target_link_libraries(AYNavigationWorldDemo PRIVATE AYNavigationLab user32 gdi32 comdlg32)
        target_compile_definitions(AYNavigationWorldDemo PRIVATE UNICODE _UNICODE NOMINMAX)
        add_custom_target(AYNavigationExampleAssets
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/examples/assets"
                "$<TARGET_FILE_DIR:AYNavigationWorldDemo>/assets" VERBATIM)
        add_dependencies(AYNavigationWorldDemo AYNavigationExampleAssets)
        if(MSVC)
            target_compile_options(AYNavigationWorldDemo PRIVATE /W4 /utf-8)
        endif()
    endif()
    add_executable(AYNavigationDemo WIN32 examples/NavigationWindow.cpp)
    target_link_libraries(AYNavigationDemo PRIVATE AYNavigationCore user32 gdi32)
    if(TARGET AYNavigationAuthoring)
        target_link_libraries(AYNavigationDemo PRIVATE AYNavigationAuthoring comdlg32)
        target_compile_definitions(AYNavigationDemo PRIVATE AY_NAVIGATION_AUTHORING=1)
    endif()
    target_compile_definitions(AYNavigationDemo PRIVATE UNICODE _UNICODE NOMINMAX)
    if(MSVC)
        target_compile_options(AYNavigationDemo PRIVATE /W4 /utf-8)
    endif()
endif()
if(BUILD_TESTING)
    if(TARGET AYNavigationPerceptionJson)
        add_executable(AYNavigationPerceptionJsonTest unittest/PerceptionJsonTest.cpp)
        target_link_libraries(AYNavigationPerceptionJsonTest PRIVATE AYNavigationPerceptionJson)
        add_test(NAME AYNavigationPerceptionJsonTest COMMAND AYNavigationPerceptionJsonTest)
    endif()
    if(TARGET AYNavigationPhysics AND AYPHYSICS_BUILD_JOLT)
        add_executable(AYNavigationPhysicsTest unittest/PhysicsTest.cpp)
        target_link_libraries(AYNavigationPhysicsTest PRIVATE AYNavigationPhysics)
        add_test(NAME AYNavigationPhysicsTest COMMAND AYNavigationPhysicsTest)
    endif()
    if(TARGET AYNavigationLab)
        add_executable(AYNavigationWorldLabTest unittest/WorldLabTest.cpp)
        target_link_libraries(AYNavigationWorldLabTest PRIVATE AYNavigationLab)
        target_compile_definitions(AYNavigationWorldLabTest PRIVATE
            AY_NAVIGATION_EXAMPLE_GRAPH="${CMAKE_CURRENT_SOURCE_DIR}/examples/assets/narrow-jump.nvg")
        add_test(NAME AYNavigationWorldLabTest COMMAND AYNavigationWorldLabTest)
    endif()
    if(TARGET AYNavigationWorldDemo)
        add_test(NAME AYNavigationWorldDemoSmoke COMMAND AYNavigationWorldDemo --smoke)
        set_tests_properties(AYNavigationWorldDemoSmoke PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 45)
    endif()
    if(TARGET AYNavigationAuthoring)
        add_executable(AYNavigationAssetTest unittest/GraphAssetTest.cpp)
        target_link_libraries(AYNavigationAssetTest PRIVATE AYNavigationAuthoring)
        add_test(NAME AYNavigationAssetTest COMMAND AYNavigationAssetTest)
    endif()
    add_executable(AYNavigationCoreTest unittest/NavigationTest.cpp)
    target_link_libraries(AYNavigationCoreTest PRIVATE AYNavigationCore)
    add_test(NAME AYNavigationCoreTest COMMAND AYNavigationCoreTest)
    add_executable(AYNavigationSpatialAwarenessTest unittest/SpatialAwarenessTest.cpp)
    target_link_libraries(AYNavigationSpatialAwarenessTest PRIVATE AYNavigationCore)
    add_test(NAME AYNavigationSpatialAwarenessTest COMMAND AYNavigationSpatialAwarenessTest)
    if(TARGET AYNavigationWorldAdapter)
        add_executable(AYNavigationWorldAdapterTest unittest/WorldAdapterTest.cpp)
        target_link_libraries(AYNavigationWorldAdapterTest PRIVATE AYNavigationWorldAdapter)
        add_test(NAME AYNavigationWorldAdapterTest COMMAND AYNavigationWorldAdapterTest)
    endif()
    if(TARGET AYNavigationDemo)
        add_test(NAME AYNavigationDemoSmoke COMMAND AYNavigationDemo --smoke)
        set_tests_properties(AYNavigationDemoSmoke PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    endif()
    if(TARGET AYNavigationBehavior)
        add_executable(AYNavigationBehaviorTest unittest/BehaviorTest.cpp)
        target_link_libraries(AYNavigationBehaviorTest PRIVATE AYNavigationBehavior)
        add_test(NAME AYNavigationBehaviorTest COMMAND AYNavigationBehaviorTest)
    endif()
    if(TARGET AYNavigationEntity)
        add_executable(AYNavigationEntityTest unittest/EntityTest.cpp)
        target_link_libraries(AYNavigationEntityTest PRIVATE AYNavigationEntity)
        add_test(NAME AYNavigationEntityTest COMMAND AYNavigationEntityTest)
    endif()
    if(NOT PROJECT_IS_TOP_LEVEL AND EXISTS "${CMAKE_SOURCE_DIR}/tools/ayapi/ayapi.ps1")
        add_test(NAME AYNavigationSdkSearch COMMAND pwsh -NoProfile -File
            "${CMAKE_CURRENT_SOURCE_DIR}/unittest/CheckSdk.ps1" -Root "${CMAKE_SOURCE_DIR}")
    endif()
endif()

add_executable(AYNavigationSpatialAwarenessExample examples/SpatialAwareness.cpp)
target_link_libraries(AYNavigationSpatialAwarenessExample PRIVATE AYNavigationCore)
